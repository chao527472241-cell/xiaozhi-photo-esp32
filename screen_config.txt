# ============================================
# 小智屏幕面板 · 设置文件
# ============================================
# 用法：改完本文件保存 → 重新编译烧录固件一次 → 之后在屏幕上点按钮切换即可，不用再编译。
# 也可以平时直接在屏幕上操作：点「主题」按钮切明暗、点「下一页」按钮换页面（断电会自动记住）。
#
# ----- 主题 -----
# 开机默认主题：light=亮色（白底黑字） / dark=暗色（黑底白字）
theme=dark

# 开机默认停在哪一页：1=时钟页  2=状态页  3=动图页
default_page=1

# ----- 时钟页 -----
# 是否显示秒：1=显示 0=不显示（显示秒更活泼，但更耗电一点点）
show_seconds=1

# ----- 状态页：实用信息开关（1=显示 0=隐藏）-----
# 你想在状态页看到哪些信息，就把对应项设成 1，不想看设成 0
show_ip=1
show_heap=1
show_uptime=1
show_state=1
show_signal=1

# ----- 动图页 -----
# 自定义标语（动图页底部显示的文字，改成你喜欢的）
slogan=你好，我是小智

# 动图切换速度（毫秒）：数字越小越快，1000=1秒。建议 600~1500
emoji_interval=900

# 动图用哪些表情（用逗号隔开，从下面挑；每个都能显示才算成功）
# 可选：😀 😊 😄 😍 🤖 👀 🌟 ❤️ 🔋 📶 ⏰ 🌈 ☀️ 🌙
emojis=😀,😊,😄,🤖,👀,🌟,🌈

# ============================================
# 说明：
# 1) 中文标语/表情需要固件里的中文与 emoji 字体支持，本固件已内置。
# 2) show_ip 需要连上 WiFi 才有地址；没连上会显示「未连接」。
# 3) 改任何一项都要重新编译一次才会写进固件；日常切换用屏幕按钮即可。
# ============================================
