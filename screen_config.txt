# 1) 时钟页底部那句标语
slogan=你好，我是小智

# 2) 配色：dark=暗色（推荐）  light=亮色
theme=dark

# 3) 开机先显示哪个界面：home=磁贴主界面  clock=大号时钟
boot=home

# 4) 壁纸开关：1=开  0=关（关掉省 flash，只有纯色背景）
#    想换成自己的照片：把图片改名成 wallpaper.jpg 放到本目录，重新编译即可。
#    建议横图（4:3 左右），会自动裁成 320x240。
wallpaper=1

# 5) 状态页显示哪几项（1=显示  0=隐藏）
show_ip=1
show_signal=1
show_heap=1
show_uptime=1
show_state=1

#    状态    → IP / WiFi 信号 / 内存 / 运行时长 / 设备状态
