# ============================================================
#  小智屏幕设置（改完要重新编译烧录一次才生效）
#  ------------------------------------------------------------
#  日常切页、换配色，直接在屏幕上点就行，不用改这个文件。
#  这里改的是"开机默认长什么样"。
# ============================================================

# 1) 时钟页底部那句标语
slogan=你好，我是小智

# 2) 配色：dark=暗色（推荐）  light=亮色
theme=dark

# 3) 开机先显示哪个界面：home=磁贴主界面  clock=大号时钟
boot=home

# 4) 壁纸开关：1=开  0=关（关掉省 flash，只有纯色背景）
#    想换成自己的照片：把图片改名成 wallpaper.jpg 放到本目录，重新编译即可。
#    建议横图（4:3 左右），会自动裁成 320x240。
wallpaper=1

# 5) 状态页显示哪几项（1=显示  0=隐藏）
show_ip=1
show_signal=1
show_heap=1
show_uptime=1
show_state=1

# 6) 板载六轴手势（QMI8658）：1=开  0=关
#    拿起板子 → 亮屏并重新计时
#    摇一摇   → 翻到下一页（主页→时钟→状态→电脑→主页，跳过设置页）
#    屏幕朝下 → 自动静音；翻回来恢复原音量
imu=1

# 7) 多久没人动就关背光（秒）：30=三十秒  0=永不熄屏
#    熄屏后点屏幕一下、或者拿起板子、摇一下都会亮起来。
#    注意：说话/聆听/回复过程中不会熄屏。
dim=30

# ------------------------------------------------------------
#  屏幕上的五个磁贴
#    小智AI  → 收起面板（说话/唤醒时面板本来也会自动让位）
#    时钟    → 大号数字时钟 + 日期 + 标语
#    状态    → 板子自己：IP / WiFi 信号 / 内存 / 运行时长 / 设备状态
#    电脑    → 电脑运行信息：CPU / 内存 / 磁盘 / 网络 / CPU 温度 / 芯片温度
#              数据由电脑端监听脚本经 USB 串口每 2 秒发过来；
#              没插电脑或脚本没跑时，这一页显示「等待电脑」。
#    设置    → 启动界面、配色、设备信息
#  空白处点一下 = 翻下一页；设置页的「< 设置」= 回主界面。
#  换配色、换启动界面后会断电记住。
# ------------------------------------------------------------
